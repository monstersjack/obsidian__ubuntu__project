下面是整理好的版本：**每个动作持续 10 秒**，并且每条命令都加了注释。  
你的方向定义保持为：

```text
ID1 = 右轮
ID2 = 左轮
小车前进 = ID1 正转，ID2 反转
小车后退 = ID1 反转，ID2 正转
```

```bash
# 重新配置 can0，设置 CAN 波特率为 1Mbps
sudo ip link set can0 down
sudo ip link set can0 type can bitrate 1000000 restart-ms 100
sudo ip link set can0 up

# 停止所有电机，确保测试从静止开始
cansend can0 032#0000000000000000
sleep 0.5

# 使能 ID1 右轮 和 ID2 左轮
# 0A = 使能
cansend can0 105#0A0A000000000000
sleep 0.2

# 设置 ID1 右轮 和 ID2 左轮为速度环模式
# 02 = 速度环
cansend can0 105#0202000000000000
sleep 0.2

# 设置 ID1 右轮 和 ID2 左轮主动反馈周期为 10ms
# 0A = 10ms 主动反馈
cansend can0 106#0A0A000000000000
sleep 0.5

# =========================
# 慢速测试：10RPM
# 10RPM = 0x0064
# -10RPM = 0xFF9C
# =========================

# 慢速前进 10RPM，持续 10 秒
# ID1 右轮：+10RPM
# ID2 左轮：-10RPM
for i in {1..500}; do cansend can0 032#0064FF9C00000000; sleep 0.02; done

# 停止所有电机，停 1 秒
cansend can0 032#0000000000000000
sleep 1

# 慢速后退 10RPM，持续 10 秒
# ID1 右轮：-10RPM
# ID2 左轮：+10RPM
for i in {1..500}; do cansend can0 032#FF9C006400000000; sleep 0.02; done

# 停止所有电机，停 1 秒
cansend can0 032#0000000000000000
sleep 1

# 原地慢速左转 10RPM，持续 10 秒
# ID1 右轮：-10RPM
# ID2 左轮：-10RPM
for i in {1..500}; do cansend can0 032#FF9CFF9C00000000; sleep 0.02; done

# 停止所有电机，停 1 秒
cansend can0 032#0000000000000000
sleep 1

# 原地慢速右转 10RPM，持续 10 秒
# ID1 右轮：+10RPM
# ID2 左轮：+10RPM
for i in {1..500}; do cansend can0 032#0064006400000000; sleep 0.02; done

# 停止所有电机，停 1 秒
cansend can0 032#0000000000000000
sleep 1

# =========================
# 中速测试：30RPM
# 30RPM = 0x012C
# -30RPM = 0xFED4
# =========================

# 中速前进 30RPM，持续 10 秒
# ID1 右轮：+30RPM
# ID2 左轮：-30RPM
for i in {1..500}; do cansend can0 032#012CFED400000000; sleep 0.02; done

# 停止所有电机，停 1 秒
cansend can0 032#0000000000000000
sleep 1

# 中速后退 30RPM，持续 10 秒
# ID1 右轮：-30RPM
# ID2 左轮：+30RPM
for i in {1..500}; do cansend can0 032#FED4012C00000000; sleep 0.02; done

# 停止所有电机，停 1 秒
cansend can0 032#0000000000000000
sleep 1

# 原地中速左转 30RPM，持续 10 秒
# ID1 右轮：-30RPM
# ID2 左轮：-30RPM
for i in {1..500}; do cansend can0 032#FED4FED400000000; sleep 0.02; done

# 停止所有电机，停 1 秒
cansend can0 032#0000000000000000
sleep 1

# 原地中速右转 30RPM，持续 10 秒
# ID1 右轮：+30RPM
# ID2 左轮：+30RPM
for i in {1..500}; do cansend can0 032#012C012C00000000; sleep 0.02; done

# 停止所有电机，停 1 秒
cansend can0 032#0000000000000000
sleep 1

# =========================
# 快速测试：60RPM
# 60RPM = 0x0258
# -60RPM = 0xFDA8
# =========================

# 快速前进 60RPM，持续 10 秒
# ID1 右轮：+60RPM
# ID2 左轮：-60RPM
for i in {1..500}; do cansend can0 032#0258FDA800000000; sleep 0.02; done

# 停止所有电机，停 1 秒
cansend can0 032#0000000000000000
sleep 1

# 快速后退 60RPM，持续 10 秒
# ID1 右轮：-60RPM
# ID2 左轮：+60RPM
for i in {1..500}; do cansend can0 032#FDA8025800000000; sleep 0.02; done

# 停止所有电机，停 1 秒
cansend can0 032#0000000000000000
sleep 1

# 原地快速左转 60RPM，持续 10 秒
# ID1 右轮：-60RPM
# ID2 左轮：-60RPM
for i in {1..500}; do cansend can0 032#FDA8FDA800000000; sleep 0.02; done

# 停止所有电机，停 1 秒
cansend can0 032#0000000000000000
sleep 1

# 原地快速右转 60RPM，持续 10 秒
# ID1 右轮：+60RPM
# ID2 左轮：+60RPM
for i in {1..500}; do cansend can0 032#0258025800000000; sleep 0.02; done

# 最终停止所有电机
cansend can0 032#0000000000000000
```
